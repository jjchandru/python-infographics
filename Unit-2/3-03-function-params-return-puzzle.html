Function accepting parameters and returning value

🧩 PUZZLE - PREDICT THE OUTPUT

def volume_of_cylinder(radius, height=5):
    pi = 3.14159
    volume = pi * radius ** 2 * height
    return volume

print(volume_of_cylinder(7, 10))

print(volume_of_cylinder(height=5, radius=14))

print(volume_of_cylinder(7))
